**Tech Stack:** Python 3.12 via `uv`, `vieneu` 3.8.x (v3 Turbo), torch 2.8.0 + cu128 (GPU venv), `voxcpm` on Colab T4, ffmpeg (libopus), plain HTML/JS MediaSource test page served by `python -m http.server`.

## Global Constraints

- All spike code is THROWAWAY and lives only under `spikes/m0/`. It is not imported by product code.
- Python for spikes: **3.12** (torch 2.8 wheels; `voxcpm` requires `>=3.10,<3.13`). System Python 3.14 is not used.
- VieNeu: v3 Turbo (`Vieneu()` default), output **48 kHz** float32 mono.
- CPU path must use fp32 (default). Do **not** pass `precision="int8"`: the i5-10500H has no AVX-512 VNNI / AVX-VNNI and int8 can produce garbled audio.
- Audio chunk format under test: standalone **WebM/Opus 48 kbps, 48 kHz, mono**, ~30 s per chunk (spec 5.6).
- No copyrighted chapter text is committed. Benchmarks fetch a chapter at runtime from `https://truyenfull.live/ga-nham-cho-dai-lao-trong-van-nien-dai/chuong-1/` (verified reachable 2026-10-13).
- Generated audio, venvs and downloaded refs are gitignored.
- All Python scripts call `utf8_console()` first (Windows console defaults to cp1252 and crashes on Vietnamese).
- Run all commands from the repo root unless a step says otherwise. Shell is Git Bash on Windows; venv interpreters are at `spikes/m0/.venv-*/Scripts/python.exe`.

```python
"""Shared helpers for the M0 spikes. THROWAWAY CODE: not part of the product."""

from __future__ import annotations

import re
import sys

SAMPLE_RATE = 48_000
DEFAULT_CHAPTER_URL = "https://truyenfull.live/ga-nham-cho-dai-lao-trong-van-nien-dai/chuong-1/"
USER_AGENT = (
    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
    "(KHTML, like Gecko) Chrome/130.0 Safari/537.36"
)

_SEO_TAIL = re.compile(r"truyen ?full.*$", re.IGNORECASE | re.DOTALL)
_DOTTED = re.compile(r"(?<!\w)((?:\w\.){2,}\w)(?!\w)")
_SENTENCE = re.compile(r".+?(?:[.!?…]+[\"”’)]*(?=\s)|$)", re.DOTALL)


def utf8_console() -> None:
    for stream in (sys.stdout, sys.stderr):
        stream.reconfigure(encoding="utf-8")


def extract_chapter_text(html: str) -> str:
    from bs4 import BeautifulSoup

    soup = BeautifulSoup(html, "lxml")
    node = soup.select_one("#chapter-c")
    if node is None:
        raise ValueError("#chapter-c not found")
    for junk in node.select('[id^="ads"], [class*="ads"]'):
        junk.decompose()
    for br in node.find_all("br"):
        br.replace_with("\n")
    text = _SEO_TAIL.sub("", node.get_text())
    lines = (line.strip() for line in text.splitlines())
    return "\n".join(line for line in lines if line)


def fetch_chapter_text(url: str) -> str:
    import httpx

    response = httpx.get(url, headers={"User-Agent": USER_AGENT}, follow_redirects=True, timeout=30)
    response.raise_for_status()
    return extract_chapter_text(response.text)


def deobfuscate(text: str) -> str:
    return _DOTTED.sub(lambda m: m.group(1).replace(".", ""), text)


def _wrap(sentence: str, max_chars: int) -> list[str]:
    if len(sentence) <= max_chars:
        return [sentence]
    parts: list[str] = []
    current = ""
    for token in re.split(r"(?<=,)\s+|\s+", sentence):
        candidate = f"{current} {token}".strip()
        if len(candidate) > max_chars and current:
            parts.append(current)
            current = token
        else:
            current = candidate
    if current:
        parts.append(current)
    return parts


def split_sentences(text: str, max_chars: int = 250) -> list[str]:
    out: list[str] = []
    for paragraph in text.split("\n"):
        paragraph = paragraph.strip()
        if not paragraph:
            continue
        for match in _SENTENCE.findall(paragraph):
            sentence = match.strip()
            if sentence:
                out.extend(_wrap(sentence, max_chars))
    return out
```

```python
"""S3: VieNeu v3 Turbo performance on this laptop. THROWAWAY."""

from __future__ import annotations

import argparse
import inspect
import json
import platform
import time
from pathlib import Path

import numpy as np

from common import (
    DEFAULT_CHAPTER_URL,
    SAMPLE_RATE,
    deobfuscate,
    fetch_chapter_text,
    split_sentences,
    utf8_console,
)


def cuda_info() -> dict:
    try:
        import torch
    except ImportError:
        return {"torch": None, "cuda": False}
    info = {"torch": torch.__version__, "cuda": torch.cuda.is_available()}
    if info["cuda"]:
        info["device"] = torch.cuda.get_device_name(0)
        info["capability"] = list(torch.cuda.get_device_capability(0))
        torch.cuda.reset_peak_memory_stats()
    return info


def peak_vram_mb() -> float | None:
    try:
        import torch
    except ImportError:
        return None
    if not torch.cuda.is_available():
        return None
    return round(torch.cuda.max_memory_allocated() / 2**20, 1)


def main() -> None:
    utf8_console()
    parser = argparse.ArgumentParser()
    parser.add_argument("--backend", choices=["auto", "onnx"], default="auto")
    parser.add_argument("--voice", default="Hải Đăng")
    parser.add_argument("--url", default=DEFAULT_CHAPTER_URL)
    parser.add_argument("--max-sentences", type=int, default=40)
    parser.add_argument("--tag", required=True, help="gpu or cpu, used in output file names")
    args = parser.parse_args()

    out_dir = Path("out")
    out_dir.mkdir(exist_ok=True)
    result: dict = {"platform": platform.platform(), "backend_arg": args.backend, **cuda_info()}

    from vieneu import Vieneu

    t0 = time.perf_counter()
    tts = Vieneu() if args.backend == "auto" else Vieneu(backend="onnx")
    result["load_s"] = round(time.perf_counter() - t0, 2)
    result["infer_signature"] = str(inspect.signature(tts.infer))
    result["infer_batch_signature"] = str(inspect.signature(tts.infer_batch))
    result["preset_voices"] = [list(v) for v in tts.list_preset_voices()]

    sentences = split_sentences(deobfuscate(fetch_chapter_text(args.url)))[: args.max_sentences]
    result["sentences"] = len(sentences)
    result["chars"] = sum(len(s) for s in sentences)

    t0 = time.perf_counter()
    tts.infer("Xin chào.", voice=args.voice)
    result["warmup_s"] = round(time.perf_counter() - t0, 2)

    audios: list[np.ndarray] = []
    t0 = time.perf_counter()
    for i, sentence in enumerate(sentences):
        audios.append(np.asarray(tts.infer(sentence, voice=args.voice), dtype=np.float32))
        if i == 0:
            result["first_sentence_s"] = round(time.perf_counter() - t0, 2)
    sequential_s = time.perf_counter() - t0
    audio_s = sum(len(a) for a in audios) / SAMPLE_RATE
    result["audio_s"] = round(audio_s, 1)
    result["sequential_s"] = round(sequential_s, 1)
    result["sequential_rtf"] = round(sequential_s / audio_s, 3)

    t0 = time.perf_counter()
    batch = tts.infer_batch(sentences, voice=args.voice)
    batch_s = time.perf_counter() - t0
    batch_audio_s = sum(len(a) for a in batch) / SAMPLE_RATE
    result["batch_s"] = round(batch_s, 1)
    result["batch_rtf"] = round(batch_s / batch_audio_s, 3)
    result["peak_vram_mb"] = peak_vram_mb()

    wav_path = out_dir / f"s3_{args.tag}.wav"
    tts.save(np.concatenate(audios), str(wav_path))
    (out_dir / f"s3_{args.tag}.json").write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding="utf-8")
    print(json.dumps({k: v for k, v in result.items() if k != "preset_voices"}, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```

"""S1-A: neutral base clips of three VieNeu presets (timbre source for VoxCPM2). THROWAWAY."""

from __future__ import annotations

import argparse
from pathlib import Path

from common import utf8_console

BASE_TEXT = "Con đường mòn dẫn lên núi phủ đầy lá vàng, gió chiều thổi qua từng rặng tre xào xạc."


def main() -> None:
    utf8_console()
    parser = argparse.ArgumentParser()
    parser.add_argument("--backend", choices=["auto", "onnx"], default="auto")
    parser.add_argument("--narrator", default="Hải Đăng")
    parser.add_argument("--male", default="Thiện Minh")
    parser.add_argument("--female", default="Ngọc Huyền")
    args = parser.parse_args()

    from vieneu import Vieneu

    tts = Vieneu() if args.backend == "auto" else Vieneu(backend="onnx")
    labels = {label for label, _ in tts.list_preset_voices()} | {vid for _, vid in tts.list_preset_voices()}
    roles = {"narrator": args.narrator, "male": args.male, "female": args.female}
    missing = [name for name in roles.values() if name not in labels]
    if missing:
        raise SystemExit(f"Unknown preset voice(s): {missing}. Available: {sorted(labels)}")

    out = Path("out/s1/base")
    out.mkdir(parents=True, exist_ok=True)
    for role, voice in roles.items():
        audio = tts.infer(BASE_TEXT, voice=voice)
        tts.save(audio, str(out / f"{role}.wav"))
        print(f"{role}: {voice} -> {out / f'{role}.wav'}")


if __name__ == "__main__":
    main()
```

"""S2: encode audio into standalone WebM/Opus chunks + manifest. THROWAWAY."""

from __future__ import annotations

import argparse
import json
import subprocess
from pathlib import Path

import numpy as np
import soundfile as sf

from common import SAMPLE_RATE, utf8_console


def sine(seconds: float, freq: float = 440.0) -> np.ndarray:
    t = np.arange(int(seconds * SAMPLE_RATE)) / SAMPLE_RATE
    return (0.3 * np.sin(2 * np.pi * freq * t)).astype(np.float32)


def encode(audio: np.ndarray, dst: Path) -> None:
    tmp = dst.with_suffix(".wav")
    sf.write(tmp, audio, SAMPLE_RATE, subtype="FLOAT")
    subprocess.run(
        ["ffmpeg", "-y", "-loglevel", "error", "-i", str(tmp),
         "-c:a", "libopus", "-b:a", "48k", "-ar", "48000", "-ac", "1", "-f", "webm", str(dst)],
        check=True,
    )
    tmp.unlink()


def make_set(name: str, audio: np.ndarray, chunk_seconds: float, root: Path) -> None:
    folder = root / name
    folder.mkdir(parents=True, exist_ok=True)
    size = int(chunk_seconds * SAMPLE_RATE)
    chunks = []
    for index, start in enumerate(range(0, len(audio), size)):
        part = audio[start : start + size]
        filename = f"chunk-{index:04d}.webm"
        encode(part, folder / filename)
        chunks.append({"url": filename, "samples": len(part), "duration": len(part) / SAMPLE_RATE})
    manifest = {"chunks": chunks, "total": len(audio) / SAMPLE_RATE}
    (folder / "manifest.json").write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    print(f"{name}: {len(chunks)} chunks, {manifest['total']:.2f}s")


def main() -> None:
    utf8_console()
    parser = argparse.ArgumentParser()
    parser.add_argument("--speech", default="out/s3_gpu.wav")
    args = parser.parse_args()
    root = Path("out/s2")
    make_set("sine", sine(30.0), 10.0, root)
    speech, rate = sf.read(args.speech, dtype="float32")
    if rate != SAMPLE_RATE:
        raise SystemExit(f"{args.speech} is {rate} Hz, expected {SAMPLE_RATE}")
    if speech.ndim > 1:
        speech = speech.mean(axis=1)
    make_set("speech", speech[: 90 * SAMPLE_RATE], 30.0, root)


if __name__ == "__main__":
    main()
```
